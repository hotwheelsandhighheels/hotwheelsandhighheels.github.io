<script>
(function() {
  const board = document.getElementById('scrap-board');
  if (!board) return;

  const scraps = board.querySelectorAll('.scrap');

  scraps.forEach(scrap => {
    let offsetX = 0;
    let offsetY = 0;
    let dragging = false;

    scrap.addEventListener('mousedown', function(e) {
      if (window.innerWidth <= 600) return;
      dragging = true;
      const rect = scrap.getBoundingClientRect();
      offsetX = e.clientX - rect.left;
      offsetY = e.clientY - rect.top;
      scrap.style.zIndex = 999;
      e.preventDefault();
    });

    document.addEventListener('mousemove', function(e) {
      if (!dragging) return;
      const boardRect = board.getBoundingClientRect();
      const x = e.clientX - boardRect.left - offsetX;
      const y = e.clientY - boardRect.top - offsetY;
      scrap.style.left = x + 'px';
      scrap.style.top = y + 'px';
    });

    document.addEventListener('mouseup', function() {
      if (dragging) {
        dragging = false;
        scrap.style.zIndex = '';
      }
    });
  });
})();
</script>
