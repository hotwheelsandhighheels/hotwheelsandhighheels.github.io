# The Millennial Epoch

Separate from the millennial age cohort, and much like the volta in poetry, an epoch references a turn in history characterized by dramatic change in perspective. The lives of millennials and those succeeding them have been shaped by overlapping technological, political, economic, environmental, and cultural transformations.

(function() {
  const legend = document.getElementById('timeline-legend');
  if (!legend) return;

  const buttons = legend.querySelectorAll('button');
  const entries = document.querySelectorAll('.timeline-entry');
  const years = document.querySelectorAll('.timeline-year');

  buttons.forEach(btn => {
    btn.addEventListener('click', function() {
      const filter = this.getAttribute('data-filter');

      buttons.forEach(b => b.classList.remove('active'));
      this.classList.add('active');

      if (filter === 'all') {
        entries.forEach(e => e.classList.remove('hidden'));
        years.forEach(y => y.classList.remove('hidden'));
        document.querySelectorAll('.timeline-clear').forEach(c => c.style.display = '');
        return;
      }

      entries.forEach(entry => {
        const categories = entry.getAttribute('data-category');
        if (categories && categories.includes(filter)) {
          entry.classList.remove('hidden');
        } else {
          entry.classList.add('hidden');
        }
      });

      years.forEach(year => {
        let next = year.nextElementSibling;
        let hasVisible = false;
        while (next && !next.classList.contains('timeline-year') && !next.classList.contains('timeline-era')) {
          if (next.classList.contains('timeline-entry') && !next.classList.contains('hidden')) {
            hasVisible = true;
            break;
          }
          next = next.nextElementSibling;
        }
        if (hasVisible) {
          year.classList.remove('hidden');
        } else {
          year.classList.add('hidden');
        }
      });

      document.querySelectorAll('.timeline-clear').forEach(c => c.style.display = '');
    });
  });
})();
</script>
